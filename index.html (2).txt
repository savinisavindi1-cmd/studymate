let currentFilter = 'all';

function renderTodos() {
    const list = document.getElementById('todo-list');
    let filtered = todos;
    
    if (currentFilter === 'completed') {
        filtered = todos.filter(t => t.completed);
    } else if (currentFilter !== 'all') {
        filtered = todos.filter(t => t.type === currentFilter && !t.completed);
    }
    
    if (filtered.length === 0) {
        list.innerHTML = `
            <div class="empty-state">
                <i class="fas fa-clipboard-list"></i>
                <p>No tasks found</p>
            </div>
        `;
        return;
    }
    
    list.innerHTML = filtered.map(todo => `
        <div class="todo-item">
            <div class="todo-check ${todo.completed ? 'checked' : ''}" onclick="toggleTodo(${todo.id})">
                ${todo.completed ? '<i class="fas fa-check"></i>' : ''}
            </div>
            <span class="todo-text ${todo.completed ? 'completed' : ''}">${todo.text}</span>
            <span class="todo-badge badge-${todo.type}">${todo.type}</span>
            <button class="todo-delete" onclick="deleteTodo(${todo.id})"><i class="fas fa-trash"></i></button>
        </div>
    `).join('');
}

function addTodo() {
    const input = document.getElementById('todo-input');
    const text = input.value.trim();
    if (!text) return;
    
    const type = text.toLowerCase().includes('assignment') ? 'assignment' : 'homework';
    todos.push({
        id: Date.now(),
        text: text,
        type: type,
        completed: false
    });
    input.value = '';
    renderTodos();
}

function toggleTodo(id) {
    const todo = todos.find(t => t.id === id);
    if (todo) {
        todo.completed = !todo.completed;
        renderTodos();
    }
}

function deleteTodo(id) {
    todos = todos.filter(t => t.id !== id);
    renderTodos();
}

function filterTodos(filter, btn) {
    currentFilter = filter;
    document.querySelectorAll('.todo-tab').forEach(t => t.classList.remove('active'));
    btn.classList.add('active');
    renderTodos();
}
